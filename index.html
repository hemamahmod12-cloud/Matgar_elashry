let xlsxLoadPromise = null; function ensureXlsx(){ if(typeof XLSX !== 'undefined') return Promise.resolve(XLSX); if(xlsxLoadPromise) return xlsxLoadPromise; xlsxLoadPromise = new Promise((resolve, reject) => { // CDN first (fast, cached); fallback to local copy in repo if CDN unavailable const CDN = 'https://cdn.jsdelivr.net/npm/xlsx/dist/xlsx.full.min.js'; const script = document.createElement('script'); script.src = CDN; script.async = true; script.onload = () => typeof XLSX !== 'undefined' ? resolve(XLSX) : tryLocal(); script.onerror = tryLocal; document.head.appendChild(script); function tryLocal(){ if(typeof XLSX !== 'undefined') return resolve(XLSX); const local = document.createElement('script'); local.src = 'src/xlsx.full.min.js'; local.async = true; local.onload = () => typeof XLSX !== 'undefined' ? resolve(XLSX) : reject(new Error('xlsx-global-missing')); local.onerror = () => reject(new Error('xlsx-load-failed')); document.head.appendChild(local); } }); return xlsxLoadPromise; } async function exportProductsCsv(){ const products = activeProducts(); if(!products.length){ toast('لا توجد منتجات متاحة للحفظ'); return; } try{ await ensureXlsx(); }catch(_e){ toast('تعذر تحميل مكوّن Excel؛ جرّب مرة أخرى'); return; } const headers = ['اسم المنتج','الباركود','الفئة','سعر الشراء','سعر البيع','الكمية','حد التنبيه','الوحدة']; const rows = products.map(p => [p.name,p.barcode||'',p.category||'عام',p.buyPrice??0,p.sellPrice??0,p.qty??0,p.minQty??0,p.unit||'قطعة']); if(typeof XLSX !== 'undefined'){ const sheet = XLSX.utils.aoa_to_sheet([headers, ...rows]); sheet['!cols'] = [{wch:28},{wch:18},{wch:18},{wch:14},{wch:14},{wch:12},{wch:14},{wch:12}]; const book = XLSX.utils.book_new(); XLSX.utils.book_append_sheet(book, sheet, 'المنتجات'); const wbout = XLSX.write(book, { bookType: 'xlsx', type: 'array' }); const blob = new Blob([wbout], { type: 'application/octet-stream' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'products.xlsx'; a.rel = 'noopener'; document.body.appendChild(a); a.click(); a.remove(); setTimeout(() => URL.revokeObjectURL(url), 1000); } else { toast('مكوّن Excel غير متوفر'); } }
